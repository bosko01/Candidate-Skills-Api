-- =======================================
-- 3. CandidateSkills (nasumično povezivanje kandidata i veština)
-- =======================================
-- Napomena: povezujemo 15 kandidata i 15 veština nasumično.

INSERT INTO CandidateSkills (CandidateId, SkillId, CreatedAt)
SELECT c.Id, s.Id, GETUTCDATE()
FROM Candidates c
JOIN Skills s ON (ABS(CHECKSUM(NEWID())) % 15) = (ABS(CHECKSUM(s.Id)) % 15);

-- Dodatni random linkovi za raznovrsnost
INSERT INTO CandidateSkills (CandidateId, SkillId, CreatedAt)
SELECT TOP 50 c.Id, s.Id, DATEADD(DAY, -ABS(CHECKSUM(NEWID()) % 365), GETUTCDATE())
FROM Candidates c
CROSS JOIN Skills s
WHERE ABS(CHECKSUM(NEWID())) % 10 = 0; -- otprilike 10% slučajeva
